## 1. 문제 분해 결과

| 하위 문제 | 입력 | 처리 | 출력 | 데이터 구조 |
|---|---|---|---|---|
| 글자 입력 | 알파벳 한 글자 | 공백 제거·소문자 변환 후 한 글자·알파벳·중복 검사, 아니면 다시 묻기 | 안내 문구 | 문자열(사용한 글자) |
| 판정·화면 | 입력 글자 | 비밀 단어에 있으면 맞춘단어, 없으면 틀린단어에 추가 | 그림, 밑줄 단어, 키보드판 | 리스트 행맨그림, 2차원 리스트 키보드판 |
| 단어·난이도 | 주제·난이도 번호 | 주제 딕셔너리에서 난이도표의 글자 수에 맞는 단어 고르기 | 글자 수, 뜻 힌트 | 딕셔너리 단어사전, 2차원 리스트 난이도표 |
| 종료·기록 | 한 판 결과 | 모두 맞힘 또는 6번 틀림이면 종료, 점수·기록 추가 | 기록표, 승패 수 | 딕셔너리 점수, 2차원 리스트 기록표 |

## 2. 프로그램 코드

전체 코드는 **08-3. 산출물 예시 – 확장 버전** 페이지와 [hangman_3_extended.py](https://cdn.jsdelivr.net/gh/nabilera1/highschool_unit3_programming@main/downloads/hangman_3_extended.py)에서 볼 수 있습니다. (행맨 그림 부분 생략)

```python
# 행맨 게임 (산출물 예시) - 딕셔너리(주제별 단어·뜻, 점수), 2차원 리스트(난이도표·키보드판·기록표), 중첩 반복, 함수
import random

행맨그림 = [r'''...''', ...]          # 0~6단계 그림 7개 (game_행맨_1_기본.py와 같음, 생략)
최대틀림 = len(행맨그림) - 1                      # 6번 틀리면 끝

# 딕셔너리 안의 딕셔너리: 주제 → {영어 단어: 한글 뜻}
단어사전 = {
    '동물': {'cat': '고양이', 'tiger': '호랑이', 'rabbit': '토끼', 'zebra': '얼룩말',
             'monkey': '원숭이', 'turtle': '거북', 'penguin': '펭귄', 'elephant': '코끼리'},
    '과일': {'apple': '사과', 'grape': '포도', 'lemon': '레몬', 'banana': '바나나',
             'cherry': '체리', 'orange': '오렌지', 'pineapple': '파인애플', 'strawberry': '딸기'},
    '정보': {'list': '리스트', 'loop': '반복', 'input': '입력', 'python': '파이썬',
             'output': '출력', 'variable': '변수', 'function': '함수', 'algorithm': '알고리즘'},
}
# 2차원 리스트: 난이도표[행] = [이름, 최소 글자 수, 최대 글자 수, 뜻 힌트 제공]
난이도표 = [['쉬움', 3, 5, True],
            ['보통', 5, 6, False],
            ['어려움', 7, 20, False]]

# 2차원 리스트: 키보드판 (행 = 키보드 줄, 열 = 글자)
키보드판 = [list('qwertyuiop'),
            list('asdfghjkl'),
            list('zxcvbnm')]
def 행맨게임소개():
    print('*' * 40)
    print(' 행맨 게임 - 주제와 난이도를 골라 보세요')
    print(f' 알파벳 한 글자씩 입력, {최대틀림}번 틀리면 끝!')
    print('*' * 40)


def 번호선택(안내문, 선택지):
    """선택지(리스트)를 번호와 함께 보여 주고, 올바른 번호가 들어올 때까지 다시 묻는다"""
    for i in range(len(선택지)):
        print(f'  {i + 1}. {선택지[i]}')
    while True:
        text = input(안내문 + ' > ').strip()
        if text.isdigit() and 1 <= int(text) <= len(선택지):
            return int(text) - 1                 # 인덱스로 돌려준다
        print(f'1~{len(선택지)} 사이의 번호를 입력하세요.')
def 비밀단어생성(주제, 난이도):
    """주제의 단어 중 난이도 글자 수에 맞는 단어를 무작위로 고른다"""
    최소, 최대 = 난이도[1], 난이도[2]
    후보 = []
    for 단어 in 단어사전[주제]:
        if 최소 <= len(단어) <= 최대:
            후보.append(단어)
    if len(후보) == 0:                           # 맞는 단어가 없으면 전체에서 고른다
        후보 = list(단어사전[주제])
    return random.choice(후보)
def 밑줄단어(비밀단어, 맞춘단어):
    """맞힌 글자는 보여 주고 나머지는 _ 로 바꾼 문자열을 돌려준다"""
    결과 = ''
    for 글자 in 비밀단어:
        if 글자 in 맞춘단어:
            결과 = 결과 + 글자 + ' '
        else:
            결과 = 결과 + '_ '
    return 결과
def 키보드출력(틀린단어, 맞춘단어):
    """2차원 리스트 키보드판을 중첩 반복문으로 출력한다 (맞힘: 대문자, 틀림: ·)"""
    for 행 in range(len(키보드판)):
        print(' ' * 행, end='')                  # 줄마다 한 칸씩 들여쓰기
        for 열 in range(len(키보드판[행])):
            글자 = 키보드판[행][열]
            if 글자 in 맞춘단어:
                print(글자.upper(), end=' ')
            elif 글자 in 틀린단어:
                print('·', end=' ')
            else:
                print(글자, end=' ')
        print()
def 게임화면생성(틀린단어, 맞춘단어, 비밀단어):
    print(행맨그림[len(틀린단어)])
    print(f'남은 기회: {최대틀림 - len(틀린단어)}번   틀린 글자: {" ".join(틀린단어)}')
    키보드출력(틀린단어, 맞춘단어)
    print()
    print('단어: ' + 밑줄단어(비밀단어, 맞춘단어))


def 알파벳한글자리턴(사용한글자):
    """공백 제거·소문자 변환 후, 사용하지 않은 알파벳 한 글자가 들어올 때까지 다시 묻는다"""
    while True:
        guess = input('알파벳 한 글자 > ').strip().lower()
        if len(guess) != 1:
            print('한 글자만 입력해 주세요.')
        elif guess not in 'abcdefghijklmnopqrstuvwxyz':
            print('알파벳(a~z)만 입력해 주세요.')
        elif guess in 사용한글자:
            print('이미 입력한 글자입니다. 다른 글자를 고르세요.')
        else:
            return guess
def 모두맞췄는지(비밀단어, 맞춘단어):
    for 글자 in 비밀단어:
        if 글자 not in 맞춘단어:
            return False
    return True
def 한판진행(주제, 난이도):
    """한 판을 진행하고 [비밀단어, 결과, 틀린 횟수]를 돌려준다"""
    비밀단어 = 비밀단어생성(주제, 난이도)
    맞춘단어, 틀린단어 = '', ''
    print(f'\n[{주제} / {난이도[0]}] {len(비밀단어)}글자 단어입니다.')
    if 난이도[3]:
        print('뜻 힌트: ' + 단어사전[주제][비밀단어])

    while True:
        게임화면생성(틀린단어, 맞춘단어, 비밀단어)
        guess = 알파벳한글자리턴(틀린단어 + 맞춘단어)
        if guess in 비밀단어:
            맞춘단어 = 맞춘단어 + guess
            if 모두맞췄는지(비밀단어, 맞춘단어):
                print(f'\n축하합니다! 정답은 {비밀단어} ({단어사전[주제][비밀단어]})')
                return [비밀단어, '승', len(틀린단어)]
        else:
            틀린단어 = 틀린단어 + guess
            if len(틀린단어) == 최대틀림:
                print(행맨그림[최대틀림])
                print(f'\n아쉬워요! 정답은 {비밀단어} ({단어사전[주제][비밀단어]})')
                return [비밀단어, '패', len(틀린단어)]
def 기록표출력(기록표):
    """2차원 리스트 기록표를 중첩 반복문으로 표처럼 출력한다"""
    제목줄 = ['판', '주제', '난이도', '단어', '결과', '틀림']
    for 행 in [제목줄] + 기록표:
        for 칸 in 행:
            print(str(칸).ljust(10), end='')
        print()


def 게임재실행():
    return input('다시 게임하실래요? (y/n) > ').strip().lower().startswith('y')
def main():
    행맨게임소개()
    점수 = {'승': 0, '패': 0}                   # 딕셔너리: 결과별 횟수
    기록표 = []                                  # 2차원 리스트: [[판, 주제, 난이도, 단어, 결과, 틀림], ...]
    주제목록 = list(단어사전.keys())
    난이도이름 = [행[0] for 행 in 난이도표]

    while True:
        주제 = 주제목록[번호선택('주제 번호', 주제목록)]
        난이도 = 난이도표[번호선택('난이도 번호', 난이도이름)]
        단어, 결과, 틀림 = 한판진행(주제, 난이도)
        점수[결과] += 1
        기록표.append([len(기록표) + 1, 주제, 난이도[0], 단어, 결과, 틀림])
        if not 게임재실행():
            break

    print('\n===== 게임 기록표 =====')
    기록표출력(기록표)
    print(f'총 {len(기록표)}판  승: {점수["승"]}  패: {점수["패"]}')


if __name__ == '__main__':
    main()
```
